   147→// ---------------------------------------------------------------------------
   148→// 02 — Instructions
   149→// ---------------------------------------------------------------------------
   150→
   151→export function InstructionsSection({ form, update }: SectionProps) {
   152→  const setSteps = (steps: StrategyFormState["steps"]) => update({ steps })
   153→
   154→  const moveStep = (index: number, direction: -1 | 1) => {
   155→    const next = [...form.steps]
   156→    const target = index + direction
   157→    if (target < 0 || target >= next.length) return
   158→    ;[next[index], next[target]] = [next[target], next[index]]
   159→    setSteps(next)
   160→  }
   161→
   162→  return (
   163→    <div className="space-y-6">
   164→      <FormCard title="Execution steps" description="The ordered “How it works” sequence.">
   165→        <div className="space-y-3">
   166→          {form.steps.map((step, i) => (
   167→            <div key={i} className="rounded-lg border border-gray-200 bg-gray-50/50 p-4">
   168→              <div className="flex items-center justify-between">
   169→                <span className="font-mono text-[10px] font-semibold uppercase tracking-[0.14em] text-gray-400">
   170→                  STEP {String(i + 1).padStart(2, "0")}
   171→                </span>
   172→                <div className="flex items-center gap-0.5">
   173→                  <Button
   174→                    type="button"
   175→                    variant="ghost"
   176→                    size="icon"
   177→                    className="size-7 text-gray-400 hover:text-gray-900"
   178→                    disabled={i === 0}
   179→                    onClick={() => moveStep(i, -1)}
   180→                    aria-label="Move step up"
   181→                  >
   182→                    <ChevronUp className="size-3.5" />
   183→                  </Button>
   184→                  <Button
   185→                    type="button"
   186→                    variant="ghost"
   187→                    size="icon"
   188→                    className="size-7 text-gray-400 hover:text-gray-900"
   189→                    disabled={i === form.steps.length - 1}
   190→                    onClick={() => moveStep(i, 1)}
   191→                    aria-label="Move step down"
   192→                  >
   193→                    <ChevronDown className="size-3.5" />
   194→                  </Button>
   195→                  <Button
   196→                    type="button"
   197→                    variant="ghost"
   198→                    size="icon"
   199→                    className="size-7 text-gray-400 hover:text-red-600"
   200→                    onClick={() => setSteps(form.steps.filter((_, j) => j !== i))}
   201→                    aria-label="Delete step"
   202→                  >
   203→                    <Trash2 className="size-3.5" />
   204→                  </Button>
   205→                </div>
   206→              </div>
   207→              <div className="mt-3 space-y-3">
   208→                <Input
   209→                  value={step.title}
   210→                  onChange={(e) =>
   211→                    setSteps(form.steps.map((s, j) => (j === i ? { ...s, title: e.target.value } : s)))
   212→                  }
   213→                  placeholder="Step title, e.g. Deposit ETH and USDC"
   214→                  className="h-9 bg-white text-sm font-medium"
   215→                />
   216→                <Textarea
   217→                  value={step.description}
   218→                  onChange={(e) =>
   219→                    setSteps(
   220→                      form.steps.map((s, j) => (j === i ? { ...s, description: e.target.value } : s))
   221→                    )
   222→                  }
   223→                  placeholder="What happens in this step and what the user should decide or watch…"
   224→                  rows={2}
   225→                  className="bg-white text-sm"
   226→                />
   227→              </div>
   228→            </div>
   229→          ))}
   230→          <Button
   231→            type="button"
   232→            variant="outline"
   233→            size="sm"
   234→            className="w-full border-dashed"
   235→            onClick={() => setSteps([...form.steps, { title: "", description: "" }])}
   236→          >
   237→            <Plus className="size-3.5" /> Add step
   238→          </Button>
   239→        </div>
   240→      </FormCard>
   241→
   242→      <div className="grid gap-6 lg:grid-cols-2">
   243→        <FormCard title="Entry conditions" description="When should someone start this strategy?">
   244→          <StringListEditor
   245→            items={form.entryConditions}
   246→            onChange={(items) => update({ entryConditions: items })}
   247→            placeholder="e.g. Market regime is sideways"
   248→            addLabel="Add entry condition"
   249→          />
   250→        </FormCard>
   251→        <FormCard title="Exit conditions" description="When should someone stop?">
   252→          <StringListEditor
   253→            items={form.exitConditions}
   254→            onChange={(items) => update({ exitConditions: items })}
   255→            placeholder="e.g. Confirmed bull-regime breakout"
   256→            addLabel="Add exit condition"
   257→          />
   258→        </FormCard>
   259→      </div>
   260→    </div>
   261→  )
   262→}

   268→export function MarketFitSection({ form, update }: SectionProps) {
   269→  const toggleRegime = (regime: Regime) => {
   270→    const regimes = form.marketFit.regimes.includes(regime)
   271→      ? form.marketFit.regimes.filter((r) => r !== regime)
   272→      : [...form.marketFit.regimes, regime]
   273→    update({ marketFit: { ...form.marketFit, regimes } })
   274→  }
   275→
   276→  return (
   277→    <FormCard
   278→      title="Market regime fit"
   279→      description="Select the regimes this strategy fits. Optional 0–100 fit scores power future AI matching."
   280→    >
   281→      <div className="space-y-2">
   282→        {REGIMES.map((regime) => {
   283→          const checked = form.marketFit.regimes.includes(regime)
   284→          return (
   285→            <div
   286→              key={regime}
   287→              className={cn(
   288→                "flex items-center gap-3 rounded-lg border p-3 transition-colors",
   289→                checked ? "border-gray-300 bg-gray-50" : "border-gray-200"
   290→              )}
   291→            >
   292→              <Checkbox
   293→                checked={checked}
   294→                onCheckedChange={() => toggleRegime(regime)}
   295→                className="size-4 data-[state=checked]:border-gray-900 data-[state=checked]:bg-gray-900"
   296→                id={`regime-${regime}`}
   297→              />
   298→              <Label
   299→                htmlFor={`regime-${regime}`}
   300→                className="flex-1 cursor-pointer text-sm font-medium text-gray-800"
   301→              >
   302→                {REGIME_LABELS[regime]}
   303→                <span className="ml-2 text-xs font-normal text-gray-400">
   304→                  {regime === "BULL" ? "trending up" : regime === "BEAR" ? "trending down" : "ranging"}
   305→                </span>
   306→              </Label>
   307→              <div className="flex items-center gap-2">
   308→                <span className="font-mono text-[9px] uppercase tracking-[0.14em] text-gray-400">
   309→                  SCORE
   310→                </span>
   311→                <Input
   312→                  value={form.marketFit.scores[regime]}
   313→                  onChange={(e) =>
   314→                    update({
   315→                      marketFit: {
   316→                        ...form.marketFit,
   317→                        scores: { ...form.marketFit.scores, [regime]: e.target.value.replace(/\D/g, "").slice(0, 3) },
   318→                      },
   319→                    })
   320→                  }
   321→                  placeholder="0–100"
   322→                  disabled={!checked}
   323→                  className="h-8 w-20 bg-white text-right font-mono text-xs"
   324→                />
   325→              </div>
   326→            </div>
   327→          )
   328→        })}
   329→      </div>
   330→
   331→      <div className="mt-5">
   332→        <Field label="WHY THIS FIT" hint="shown on the detail page under Market Fit">
   333→          <Textarea
   334→            value={form.marketFit.explanation}
   335→            onChange={(e) =>
   336→              update({ marketFit: { ...form.marketFit, explanation: e.target.value } })
   337→            }
   338→            placeholder="Explain why the strategy suits these regimes (and not the others)…"
   339→            rows={4}
   340→            className="text-sm"
   341→          />
   342→        </Field>
   343→      </div>
   344→    </FormCard>
   345→  )
   346→}

   377→function AssetRoleGroup({
   378→  label,
   379→  hint,
   380→  role,
   381→  form,
   382→  update,
   383→  list,
   384→}: {
   385→  label: string
   386→  hint: string
   387→  role: "depositAssetIds" | "exposureAssetIds" | "rewardAssetIds"
   388→  form: StrategyFormState
   389→  update: (patch: Partial<StrategyFormState>) => void
   390→  list: { id: string; symbol: string; name: string; category: string | null; iconUrl: string | null; active: boolean }[]
   391→}) {
   392→  return (
   393→    <FormCard title={label} description={hint}>
   394→      <div className="flex flex-wrap gap-2">
   395→        {list.map((a) => {
   396→          const selected = form[role].includes(a.id)
   397→          return (
   398→            <button
   399→              key={a.id}
   400→              type="button"
   401→              onClick={() =>
   402→                update({
   403→                  [role]: selected ? form[role].filter((x) => x !== a.id) : [...form[role], a.id],
   404→                } as Partial<StrategyFormState>)
   405→              }
   406→              className={cn(
   407→                "inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1.5 text-xs font-medium transition-all",
   408→                selected
   409→                  ? "border-gray-900 bg-gray-900 text-white shadow-sm"
   410→                  : "border-gray-200 bg-white text-gray-600 hover:border-gray-400"
   411→              )}
   412→            >
   413→              <AssetIcon symbol={a.symbol} category={a.category} iconUrl={a.iconUrl} size={16} />
   414→              <span className="font-mono">{a.symbol}</span>
   415→            </button>
   416→          )
   417→        })}
   418→        {list.length === 0 && (
   419→          <p className="text-xs text-gray-400">No assets yet — create them under Admin → Assets.</p>
   420→        )}
   421→      </div>
   422→    </FormCard>
   423→  )

   930→function StringListEditor({

   932→  onChange,
   933→  placeholder,
