    54→
    55→  const [activeSection, setActiveSection] = useState<string>("overview")
    56→
    57→  // Scrollspy for the record index.
    58→  useEffect(() => {
    59→    if (!strategy) return
    60→    const observer = new IntersectionObserver(
    61→      (entries) => {
    62→        for (const entry of entries) {
    63→          if (entry.isIntersecting) setActiveSection(entry.target.id)
    64→        }
    65→      },
    66→      { rootMargin: "-20% 0px -70% 0px" }
    67→    )
    68→    for (const section of SECTIONS) {
    69→      const element = document.getElementById(section.id)
    70→      if (element) observer.observe(element)
    71→    }
    72→    return () => observer.disconnect()
    73→  }, [strategy])
    74→
    75→  const uniqueAssets = useMemo(() => {
    76→    if (!strategy) return []
    77→    return [
    78→      ...new Map(
    79→        [...strategy.depositAssets, ...strategy.exposureAssets].map((a) => [a.symbol, a])
    80→      ).values(),
    81→    ]
    82→  }, [strategy])
    83→
    84→  if (isLoading) {
    85→    return (
    86→      <div className="mx-auto max-w-[1380px] animate-pulse px-4 py-16 sm:px-8">
    87→        <div className="h-4 w-40 bg-white/5" />
    88→        <div className="mt-8 h-14 w-2/3 bg-white/5" />
    89→        <div className="mt-10 h-64 bg-white/[0.03]" />
    90→      </div>
    91→    )
    92→  }
    93→
    94→  if (isError || !strategy) {
    95→    return (
    96→      <div className="mx-auto max-w-[1380px] px-4 py-16 sm:px-8">
    97→        <div className="flex flex-col items-center border border-white/10 bg-arc-surface px-6 py-24 text-center">
    98→          <p className="arc-mono text-arc-red">RECORD NOT FOUND</p>
    99→          <p className="mt-3 max-w-md text-[14px] text-arc-muted">
   100→            This record does not exist in the archive, or it has been withdrawn.
   101→          </p>
   102→          <Link
   103→            href={urls.allRecords()}
   104→            className="arc-mono mt-8 flex h-11 items-center border border-white/15 px-6 text-arc-text transition-colors hover:border-white/40"
   105→          >
   106→            ← BACK TO ALL RECORDS
   107→          </Link>
   108→        </div>
   109→      </div>
   110→    )
   111→  }
   112→
   113→  return (
   114→    <article>
   115→      <div className="mx-auto max-w-[1380px] px-4 sm:px-8">
   116→        {/* Non-published banner */}
   117→        {strategy.status !== "PUBLISHED" && (
   118→          <div
   119→            className={cn(
   120→              "mt-6 flex items-center gap-3 border px-4 py-3",
   121→              strategy.status === "DRAFT"
   122→                ? "border-arc-amber/40 bg-arc-amber/10 text-arc-amber"
   123→                : "border-white/15 bg-white/5 text-arc-muted"
   124→            )}
   125→          >
   126→            <span className="arc-mono">
   127→              {strategy.status === "DRAFT" ? "DRAFT RECORD — NOT YET PUBLISHED" : "ARCHIVED RECORD — RETAINED FOR REFERENCE"}
   128→            </span>
   129→          </div>
   130→        )}
   131→
   132→        {/* Record header */}
   133→        <header className="border-b border-white/10 py-10 sm:py-14">
   134→          <div className="flex flex-wrap items-center gap-4">
   135→            <Link
   136→              href={urls.allRecords()}
   137→              className="arc-mono flex items-center gap-2 text-arc-muted transition-colors hover:text-arc-text"
   138→            >
   139→              <ArrowLeft className="size-3.5" aria-hidden /> ALL RECORDS
   140→            </Link>
   141→            <span className="arc-mono text-arc-dim">ARCHIVE RECORD / {strategy.strategyId}</span>
   142→          </div>
   143→
   144→          <h1 className="mt-6 max-w-4xl font-serif text-[clamp(2.4rem,5.5vw,4.4rem)] font-light leading-[1.02] tracking-[-0.02em] text-arc-text">
   145→            {strategy.name}
   146→          </h1>
   147→
   148→          <p className="arc-mono mt-5 text-[11px] text-arc-dim">
   149→            {(objectiveLabels(strategy).join(" / ") || strategy.type).toUpperCase()} ·{" "}
   150→            {strategy.type.toUpperCase()}

   242→            <Link href={urls.explore({ step: "market" })} className="arc-mono text-arc-muted transition-colors hover:text-arc-text">
   243→              NEW RETRIEVAL →
   244→            </Link>
   245→            <Link href={urls.allRecords()} className="arc-mono text-arc-muted transition-colors hover:text-arc-text">

   247→            </Link>
   248→          </div>
   249→        </div>
   250→      </div>
   251→    </article>
   252→  )
   253→}

   270→                      "-ml-px flex items-center gap-3 border-l-2 py-2 pl-4 transition-colors",

   288→        aria-label="Record sections"
   289→        className="arc-scroll -mx-4 flex gap-1.5 overflow-x-auto px-4 pb-1 lg:hidden"
   290→      >
   291→        {SECTIONS.map((section) => {
   292→          const isActive = section.id === active
   293→          return (
   294→            <a
   295→              key={section.id}
   296→              href={`#${section.id}`}
   297→              className={cn(
   298→                "arc-mono shrink-0 border px-3 py-2 text-[10px] transition-colors",

   467→function MarketFitSection({ strategy }: { strategy: StrategyDTO }) {
   468→  const regimes: Regime[] = ["BULL", "SIDEWAYS", "BEAR"]
   469→  const scores = strategy.marketFit?.scores ?? {}
   470→  const declared = strategy.marketFit?.regimes ?? []
   471→  const secondary = strategy.marketFit?.secondaryRegimes ?? []
   472→  const secondaryScores = strategy.marketFit?.secondaryScores ?? {}
   473→
   474→  return (
   475→    <section id="market-fit" aria-label="Market fit">
   476→      <SectionRule code="04" title="Market Fit" />
   477→      <div className="mt-8 max-w-2xl space-y-5">
   478→        {regimes.map((regime) => {
   479→          const score = scores[regime]
   480→          const active = declared.includes(regime)
   481→          return (
   482→            <div key={regime}>
   483→              <div className="flex items-baseline justify-between gap-4">
   484→                <span
   485→                  className={cn(
   486→                    "arc-mono text-[11px]",
   487→                    active ? REGIME_TEXT[regime] : "text-arc-dim"
   488→                  )}
   489→                >
   490→                  {REGIME_ARROWS[regime]} {regime}
   491→                  {!active && <span className="ml-2 text-arc-dim/70">NOT DESIGNED</span>}
   492→                </span>
   493→                <span className="font-mono text-[13px] tabular-nums text-arc-text">
   494→                  {typeof score === "number" ? score : "—"}
   495→                </span>
   496→              </div>
   497→              <div className="mt-2 h-[3px] w-full bg-white/10" aria-hidden>
   498→                <span
   499→                  className={cn(
   500→                    "block h-full transition-all",
   501→                    active
   502→                      ? regime === "BULL"
   503→                        ? "bg-arc-green"
   504→                        : regime === "SIDEWAYS"
   505→                          ? "bg-arc-amber"
   506→                          : "bg-arc-red"
   507→                      : "bg-white/15"
   508→                  )}
   509→                  style={{ width: `${typeof score === "number" ? score : 4}%` }}
   510→                />
   511→              </div>
   512→            </div>
   513→          )
   514→        })}
   515→      </div>
   516→      {strategy.marketFit?.explanation && (
   517→        <p className="mt-8 max-w-2xl text-[14px] leading-[1.75] text-arc-muted">
   518→          {strategy.marketFit.explanation}
   519→        </p>
   520→      )}
   521→
   522→      {/* Level-2: only the market phases this record actually declares. */}
   523→      {secondary.length > 0 && (
   524→        <div className="mt-10 max-w-2xl border-t border-white/10 pt-6">
   525→          <p className="arc-mono text-[11px] text-arc-muted">
   526→            <span className="mr-2 inline-block size-1.5 rounded-full bg-arc-green align-middle" aria-hidden />
   527→            BEST DURING — MARKET PHASES
