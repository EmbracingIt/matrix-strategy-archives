   151→   151→export function InstructionsSection({ form, update }: SectionProps) {
   152→   152→  const setSteps = (steps: StrategyFormState["steps"]) => update({ steps })
   153→   153→
   154→   154→  const moveStep = (index: number, direction: -1 | 1) => {
   155→   155→    const next = [...form.steps]
   156→   156→    const target = index + direction
   157→   157→    if (target < 0 || target >= next.length) return
   158→   158→    ;[next[index], next[target]] = [next[target], next[index]]
   159→   159→    setSteps(next)
   160→   160→  }
   161→   161→
   162→   162→  return (
   163→   163→    <div className="space-y-6">
   164→   164→      <FormCard title="Execution steps" description="The ordered “How it works” sequence.">
   165→   165→        <div className="space-y-3">
   166→   166→          {form.steps.map((step, i) => (
   167→   167→            <div key={i} className="rounded-lg border border-gray-200 bg-gray-50/50 p-4">
   168→   168→              <div className="flex items-center justify-between">
   169→   169→                <span className="font-mono text-[10px] font-semibold uppercase tracking-[0.14em] text-gray-400">
   170→   170→                  STEP {String(i + 1).padStart(2, "0")}
   171→   171→                </span>
   172→   172→                <div className="flex items-center gap-0.5">
   173→   173→                  <Button
   174→   174→                    type="button"
   175→   175→                    variant="ghost"
   176→   176→                    size="icon"
   177→   177→                    className="size-7 text-gray-400 hover:text-gray-900"
   178→   178→                    disabled={i === 0}
   179→   179→                    onClick={() => moveStep(i, -1)}
   180→   180→                    aria-label="Move step up"
   181→   181→                  >
   182→   182→                    <ChevronUp className="size-3.5" />
   183→   183→                  </Button>
   184→   184→                  <Button
   185→   185→                    type="button"
   186→   186→                    variant="ghost"
   187→   187→                    size="icon"
   188→   188→                    className="size-7 text-gray-400 hover:text-gray-900"
   189→   189→                    disabled={i === form.steps.length - 1}
   190→   190→                    onClick={() => moveStep(i, 1)}
   191→   191→                    aria-label="Move step down"
   192→   192→                  >
   193→   193→                    <ChevronDown className="size-3.5" />
   194→   194→                  </Button>
   195→   195→                  <Button
   196→   196→                    type="button"
   197→   197→                    variant="ghost"
   198→   198→                    size="icon"
   199→   199→                    className="size-7 text-gray-400 hover:text-red-600"
   200→   200→                    onClick={() => setSteps(form.steps.filter((_, j) => j !== i))}
   201→   201→                    aria-label="Delete step"
   202→   202→                  >
   203→   203→                    <Trash2 className="size-3.5" />
   204→   204→                  </Button>
   205→   205→                </div>
   206→   206→              </div>
   207→   207→              <div className="mt-3 space-y-3">
   208→   208→                <Input
   209→   209→                  value={step.title}
   210→   210→                  onChange={(e) =>
   211→   211→                    setSteps(form.steps.map((s, j) => (j === i ? { ...s, title: e.target.value } : s)))
   212→   212→                  }
   213→   213→                  placeholder="Step title, e.g. Deposit ETH and USDC"
   214→   214→                  className="h-9 bg-white text-sm font-medium"
   215→   215→                />
   216→   216→                <Textarea
   217→   217→                  value={step.description}
   218→   218→                  onChange={(e) =>
   219→   219→                    setSteps(
   220→   220→                      form.steps.map((s, j) => (j === i ? { ...s, description: e.target.value } : s))
   221→   221→                    )
   222→   222→                  }
   223→   223→                  placeholder="What happens in this step and what the user should decide or watch…"
   224→   224→                  rows={2}
   225→   225→                  className="bg-white text-sm"
   226→   226→                />
   227→   227→              </div>
   228→   228→            </div>
   229→   229→          ))}
   230→   230→          <Button
   231→   231→            type="button"
   232→   232→            variant="outline"
   233→   233→            size="sm"
   234→   234→            className="w-full border-dashed"
   235→   235→            onClick={() => setSteps([...form.steps, { title: "", description: "" }])}
   236→   236→          >
   237→   237→            <Plus className="size-3.5" /> Add step
   238→   238→          </Button>
   239→   239→        </div>
   240→   240→      </FormCard>
   241→   241→
   242→   242→      <div className="grid gap-6 lg:grid-cols-2">
   243→   243→        <FormCard title="Entry conditions" description="When should someone start this strategy?">
   244→   244→          <StringListEditor
   245→   245→            items={form.entryConditions}
   246→   246→            onChange={(items) => update({ entryConditions: items })}
   247→   247→            placeholder="e.g. Market regime is sideways"
   248→   248→            addLabel="Add entry condition"
   249→   249→          />
   250→   250→        </FormCard>
   251→   251→        <FormCard title="Exit conditions" description="When should someone stop?">
   252→   252→          <StringListEditor
   253→   253→            items={form.exitConditions}
   254→   254→            onChange={(items) => update({ exitConditions: items })}
   255→   255→            placeholder="e.g. Confirmed bull-regime breakout"
   256→   256→            addLabel="Add exit condition"
   257→   257→          />
   258→   258→        </FormCard>
   259→   259→      </div>
   260→   260→    </div>
   261→   261→  )
   262→   262→}

   268→   268→export function MarketFitSection({ form, update }: SectionProps) {

   280→   280→    >

   282→   282→        {REGIMES.map((regime) => {
   283→   283→          const checked = form.marketFit.regimes.includes(regime)
   284→   284→          return (
   285→   285→            <div
   286→   286→              key={regime}
   287→   287→              className={cn(

   322→   322→                  disabled={!checked}

   324→   324→                />
   325→   325→              </div>
   326→   326→            </div>
   327→   327→          )

   329→   329→      </div>
   330→   330→
   331→   331→      <div className="mt-5">
