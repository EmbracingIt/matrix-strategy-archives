   165→   165→
   166→   166→export function InstructionsSection({ form, update }: SectionProps) {
   167→   167→  const setSteps = (steps: StrategyFormState["steps"]) => update({ steps })
   168→   168→
   169→   169→  const moveStep = (index: number, direction: -1 | 1) => {
   170→   170→    const next = [...form.steps]
   171→   171→    const target = index + direction
   172→   172→    if (target < 0 || target >= next.length) return
   173→   173→    ;[next[index], next[target]] = [next[target], next[index]]
   174→   174→    setSteps(next)
   175→   175→  }
   176→   176→
   177→   177→  return (
   178→   178→    <SectionSurface
   179→   179→      title="Instructions"
   180→   180→      description="The step-by-step execution flow shown on the strategy detail page."
   181→   181→    >
   182→   182→      <div className="space-y-3">
   183→   183→        {form.steps.map((step, i) => (
   184→   184→          <div key={i} className="rounded-lg border border-gray-200 bg-gray-50/60">
   185→   185→            <div className="flex items-center justify-between border-b border-gray-200/70 px-4 py-2">
   186→   186→              <span className="font-mono text-xs font-semibold text-gray-400">
   187→   187→                {String(i + 1).padStart(2, "0")}
   188→   188→              </span>
   189→   189→              <div className="flex items-center gap-0.5">
   190→   190→                <Button
   191→   191→                  type="button"
   192→   192→                  variant="ghost"
   193→   193→                  size="icon"
   194→   194→                  className="size-8 text-gray-400 hover:text-gray-900"
   195→   195→                  disabled={i === 0}
   196→   196→                  onClick={() => moveStep(i, -1)}
   197→   197→                  aria-label="Move step up"
   198→   198→                >
   199→   199→                  <ArrowUp className="size-4" />
   200→   200→                </Button>
   201→   201→                <Button
   202→   202→                  type="button"
   203→   203→                  variant="ghost"
   204→   204→                  size="icon"
   205→   205→                  className="size-8 text-gray-400 hover:text-gray-900"
   206→   206→                  disabled={i === form.steps.length - 1}
   207→   207→                  onClick={() => moveStep(i, 1)}
   208→   208→                  aria-label="Move step down"
   209→   209→                >
   210→   210→                  <ArrowDown className="size-4" />
   211→   211→                </Button>
   212→   212→                <Button
   213→   213→                  type="button"
   214→   214→                  variant="ghost"
   215→   215→                  size="icon"
   216→   216→                  className="size-8 text-gray-400 hover:bg-red-50 hover:text-red-600"
   217→   217→                  onClick={() => setSteps(form.steps.filter((_, j) => j !== i))}
   218→   218→                  aria-label="Delete step"
   219→   219→                >
   220→   220→                  <Trash2 className="size-4" />
   221→   221→                </Button>
   222→   222→              </div>
   223→   223→            </div>
   224→   224→            <div className="space-y-4 p-4">
   225→   225→              <Field label="Step title">
   226→   226→                <Input
   227→   227→                  value={step.title}
   228→   228→                  onChange={(e) =>
   229→   229→                    setSteps(form.steps.map((s, j) => (j === i ? { ...s, title: e.target.value } : s)))
   230→   230→                  }
   231→   231→                  placeholder="e.g. Deposit ETH and USDC"
   232→   232→                  className="h-10 bg-white text-sm font-medium"
   233→   233→                />
   234→   234→              </Field>
   235→   235→              <Field label="Description" hint="What happens in this step and what to decide or watch.">
   236→   236→                <Textarea
   237→   237→                  value={step.description}
   238→   238→                  onChange={(e) =>
   239→   239→                    setSteps(
   240→   240→                      form.steps.map((s, j) => (j === i ? { ...s, description: e.target.value } : s))
   241→   241→                    )
   242→   242→                  }
   243→   243→                  placeholder="Details for this step…"
   244→   244→                  rows={3}
   245→   245→                  className="min-h-[80px] bg-white text-sm"
   246→   246→                />
   247→   247→              </Field>
   248→   248→            </div>
   249→   249→          </div>
   250→   250→        ))}
   251→   251→        <Button
   252→   252→          type="button"
   253→   253→          variant="outline"
   254→   254→          size="sm"
   255→   255→          className="h-9 w-full border-dashed text-xs text-gray-600"
   256→   256→          onClick={() => setSteps([...form.steps, { title: "", description: "" }])}
   257→   257→        >
   258→   258→          <Plus className="size-3.5" /> Add step
   259→   259→        </Button>
   260→   260→      </div>
   261→   261→
   262→   262→      <div className="mt-8 border-t border-gray-100 pt-7">
   263→   263→        <GroupHeading
   264→   264→          title="Entry conditions"
   265→   265→          description="When should someone start this strategy?"
   266→   266→        />
   267→   267→        <StringListEditor
   268→   268→          items={form.entryConditions}
   269→   269→          onChange={(items) => update({ entryConditions: items })}
   270→   270→          placeholder="e.g. Market regime is Bear or Sideways"
   271→   271→          addLabel="Add condition"
   272→   272→        />
   273→   273→      </div>
   274→   274→
   275→   275→      <div className="mt-7 border-t border-gray-100 pt-7">
   276→   276→        <GroupHeading
   277→   277→          title="Exit conditions"
   278→   278→          description="When should someone stop or unwind it?"
   279→   279→        />

   282→   282→          onChange={(items) => update({ exitConditions: items })}
   283→   283→          placeholder="e.g. Confirmed bull-regime breakout"
   284→   284→          addLabel="Add condition"
   285→   285→        />
   286→   286→      </div>
   287→   287→    </SectionSurface>
   288→   288→  )
   289→   289→}
