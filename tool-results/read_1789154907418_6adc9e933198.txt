   162→// ---------------------------------------------------------------------------
   163→// 02 — Instructions
   164→// ---------------------------------------------------------------------------
   165→
   166→export function InstructionsSection({ form, update }: SectionProps) {
   167→  const setSteps = (steps: StrategyFormState["steps"]) => update({ steps })
   168→
   169→  const moveStep = (index: number, direction: -1 | 1) => {
   170→    const next = [...form.steps]
   171→    const target = index + direction
   172→    if (target < 0 || target >= next.length) return
   173→    ;[next[index], next[target]] = [next[target], next[index]]
   174→    setSteps(next)
   175→  }
   176→
   177→  return (
   178→    <SectionSurface
   179→      title="Instructions"
   180→      description="The step-by-step execution flow shown on the strategy detail page."
   181→    >
   182→      <div className="space-y-3">
   183→        {form.steps.map((step, i) => (
   184→          <div key={i} className="rounded-lg border border-gray-200 bg-gray-50/60">
   185→            <div className="flex items-center justify-between border-b border-gray-200/70 px-4 py-2">
   186→              <span className="font-mono text-xs font-semibold text-gray-400">
   187→                {String(i + 1).padStart(2, "0")}
   188→              </span>
   189→              <div className="flex items-center gap-0.5">
   190→                <Button
   191→                  type="button"
   192→                  variant="ghost"
   193→                  size="icon"
   194→                  className="size-8 text-gray-400 hover:text-gray-900"
   195→                  disabled={i === 0}
   196→                  onClick={() => moveStep(i, -1)}
   197→                  aria-label="Move step up"
   198→                >
   199→                  <ArrowUp className="size-4" />
   200→                </Button>
   201→                <Button
   202→                  type="button"
   203→                  variant="ghost"
   204→                  size="icon"
   205→                  className="size-8 text-gray-400 hover:text-gray-900"
   206→                  disabled={i === form.steps.length - 1}
   207→                  onClick={() => moveStep(i, 1)}
   208→                  aria-label="Move step down"
   209→                >
   210→                  <ArrowDown className="size-4" />
   211→                </Button>
   212→                <Button
   213→                  type="button"
   214→                  variant="ghost"
   215→                  size="icon"
   216→                  className="size-8 text-gray-400 hover:bg-red-50 hover:text-red-600"
   217→                  onClick={() => setSteps(form.steps.filter((_, j) => j !== i))}
   218→                  aria-label="Delete step"
   219→                >
   220→                  <Trash2 className="size-4" />
   221→                </Button>
   222→              </div>
   223→            </div>
   224→            <div className="space-y-4 p-4">
   225→              <Field label="Step title">
   226→                <Input
   227→                  value={step.title}
   228→                  onChange={(e) =>
   229→                    setSteps(form.steps.map((s, j) => (j === i ? { ...s, title: e.target.value } : s)))
   230→                  }
   231→                  placeholder="e.g. Deposit ETH and USDC"
   232→                  className="h-10 bg-white text-sm font-medium"
   233→                />
   234→              </Field>
   235→              <Field label="Description" hint="What happens in this step and what to decide or watch.">
   236→                <Textarea
   237→                  value={step.description}
   238→                  onChange={(e) =>
   239→                    setSteps(
   240→                      form.steps.map((s, j) => (j === i ? { ...s, description: e.target.value } : s))
   241→                    )
   242→                  }
   243→                  placeholder="Details for this step…"
   244→                  rows={3}
   245→                  className="min-h-[80px] bg-white text-sm"
   246→                />
   247→              </Field>
   248→            </div>
   249→          </div>
   250→        ))}
   251→        <Button
   252→          type="button"
   253→          variant="outline"
   254→          size="sm"
   255→          className="h-9 w-full border-dashed text-xs text-gray-600"
   256→          onClick={() => setSteps([...form.steps, { title: "", description: "" }])}
   257→        >
   258→          <Plus className="size-3.5" /> Add step
   259→        </Button>
   260→      </div>
   261→
   262→      <div className="mt-8 border-t border-gray-100 pt-7">
   263→        <GroupHeading
   264→          title="Entry conditions"
   265→          description="When should someone start this strategy?"
   266→        />
   267→        <StringListEditor
   268→          items={form.entryConditions}
   269→          onChange={(items) => update({ entryConditions: items })}
   270→          placeholder="e.g. Market regime is Bear or Sideways"
   271→          addLabel="Add condition"
   272→        />
   273→      </div>
   274→
   275→      <div className="mt-7 border-t border-gray-100 pt-7">
   276→        <GroupHeading
   277→          title="Exit conditions"
   278→          description="When should someone stop or unwind it?"
   279→        />
   280→        <StringListEditor
   281→          items={form.exitConditions}
   282→          onChange={(items) => update({ exitConditions: items })}
   283→          placeholder="e.g. Confirmed bull-regime breakout"
   284→          addLabel="Add condition"
   285→        />
   286→      </div>
   287→    </SectionSurface>
   288→  )
   289→}

   295→export function MarketFitSection({ form, update }: SectionProps) {
   296→  const toggleRegime = (regime: Regime) => {
   297→    const regimes = form.marketFit.regimes.includes(regime)
   298→      ? form.marketFit.regimes.filter((r) => r !== regime)
   299→      : [...form.marketFit.regimes, regime]
   300→    update({ marketFit: { ...form.marketFit, regimes } })
   301→  }
   302→
   303→  const selectedRegimes = REGIMES.filter((r) => form.marketFit.regimes.includes(r))
   304→
   305→  return (
   306→    <SectionSurface
   307→      title="Market Fit"
   308→      description="Which market regimes this strategy is designed for. Optional 0–100 fit scores power future AI matching."
   309→    >
   310→      <div className="grid gap-3 sm:grid-cols-3">
   311→        {REGIMES.map((regime) => {
   312→          const selected = form.marketFit.regimes.includes(regime)
   313→          return (
   314→            <button
   315→              key={regime}
   316→              type="button"
   317→              onClick={() => toggleRegime(regime)}
   318→              aria-pressed={selected}
   319→              className={cn(
   320→                "rounded-lg border p-4 text-left transition-colors",
   321→                selected
   322→                  ? "border-gray-900 bg-gray-50"
   323→                  : "border-gray-200 bg-white hover:border-gray-400"
   324→              )}
   325→            >
   326→              <div className="flex items-center justify-between">
   327→                <span className="text-sm font-semibold text-gray-900">
   328→                  {REGIME_LABELS[regime]}
   329→                </span>
   330→                <span
   331→                  className={cn(
   332→                    "flex size-4 items-center justify-center rounded-full border transition-colors",
   333→                    selected
   334→                      ? "border-gray-900 bg-gray-900 text-white"
   335→                      : "border-gray-300 text-transparent"
   336→                  )}
   337→                >
   338→                  <Check className="size-3" strokeWidth={3} />
   339→                </span>
   340→              </div>
   341→              <p className="mt-1 text-xs text-gray-500">{REGIME_HINTS[regime]}</p>
   342→            </button>
   343→          )
   344→        })}
   345→      </div>
   346→
   347→      {selectedRegimes.length > 0 && (
   348→        <div className="mt-7">
   349→          <GroupHeading title="Fit scores" description="0–100 — how well the strategy fits each selected regime." />
   350→          <div className="max-w-md space-y-3">
   351→            {selectedRegimes.map((regime) => (
   352→              <div key={regime} className="flex items-center gap-4">
   353→                <span className="w-24 shrink-0 text-sm font-medium text-gray-700">
   354→                  {REGIME_LABELS[regime]}
   355→                </span>
   356→                <Input
   357→                  value={form.marketFit.scores[regime]}
   358→                  onChange={(e) =>
   359→                    update({
   360→                      marketFit: {
   361→                        ...form.marketFit,
   362→                        scores: {
   363→                          ...form.marketFit.scores,
   364→                          [regime]: e.target.value.replace(/\D/g, "").slice(0, 3),
   365→                        },
   366→                      },
   367→                    })
   368→                  }
   369→                  placeholder="0–100"
   370→                  inputMode="numeric"
   371→                  className="h-10 w-24 bg-white text-right font-mono text-sm"
   372→                />
   373→              </div>
   374→            ))}
   375→          </div>
   376→        </div>
   377→      )}
   378→
   379→      <div className="mt-8 border-t border-gray-100 pt-7">
   380→        <Field
   381→          label="Why does this strategy fit these regimes?"
   382→          hint="Shown on the detail page under Market Fit."
   383→        >
   384→          <Textarea
   385→            value={form.marketFit.explanation}
   386→            onChange={(e) =>
   387→              update({ marketFit: { ...form.marketFit, explanation: e.target.value } })
   388→            }
   389→            placeholder="Explain why the strategy suits these regimes (and not the others)…"
   390→            rows={4}
   391→            className="min-h-[128px] text-sm leading-relaxed"
   392→          />
   393→        </Field>
   394→      </div>
   395→    </SectionSurface>
   396→  )
   397→}

   403→function AssetRoleGroup({
   404→  label,
   405→  hint,
   406→  role,
   407→  form,
   408→  update,
   409→  list,
   410→}: {
   411→  label: string
   412→  hint: string
   413→  role: "depositAssetIds" | "exposureAssetIds" | "rewardAssetIds"
   414→  form: StrategyFormState
   415→  update: (patch: Partial<StrategyFormState>) => void
   416→  list: { id: string; symbol: string; name: string; category: string | null; iconUrl: string | null; active: boolean }[]
   417→}) {
   418→  const toggle = (id: string) => {
   419→    const selected = form[role].includes(id)
   420→    update({
   421→      [role]: selected ? form[role].filter((x) => x !== id) : [...form[role], id],
   422→    } as Partial<StrategyFormState>)
   423→  }
   424→
   425→  return (
   426→    <div>
   427→      <GroupHeading title={label} description={hint} />
   428→      <SearchMultiSelect
   429→        options={list.map((a) => ({
   430→          id: a.id,
   431→          label: a.symbol,
   432→          sub: a.name,
   433→          icon: <AssetIcon symbol={a.symbol} category={a.category} iconUrl={a.iconUrl} size={18} />,
   434→        }))}
   435→        selected={form[role]}
   436→        onToggle={toggle}
   437→        placeholder="Search assets…"
   438→        searchPlaceholder="Search by symbol or name…"
   439→        emptyText={list.length === 0 ? "No assets yet — create them under Assets." : "No matches."}
   440→      />
   441→    </div>
   442→  )
   443→}

   583→export function RiskSection({ form, update }: SectionProps) {
   584→  const risk = form.risk
   585→  const setRisk = (patch: Partial<StrategyFormState["risk"]>) =>
   586→    update({ risk: { ...risk, ...patch } })
   587→
   588→  const riskOptions = RISK_LEVELS.map((v) => ({ value: v, label: RISK_LABELS[v] }))
   589→  const exposureOptions = EXPOSURE_LEVELS.map((v) => ({ value: v, label: EXPOSURE_LABELS[v] }))
   590→
   591→  return (
   592→    <SectionSurface
   593→      title="Risk"
   594→      description="Structured risk classification — used by the public filters and future AI matching."
   595→    >
   596→      <div className="grid gap-6 sm:grid-cols-2">
   597→        <RiskSelect
   598→          label="Overall risk"
   599→          value={risk.overallRisk}
   600→          options={riskOptions}
   601→          onChange={(v) => setRisk({ overallRisk: v as RiskLevel })}
   602→        />
   603→        <RiskSelect
   604→          label="Liquidation exposure"
   605→          value={risk.liquidationExposure}
   606→          options={exposureOptions}
   607→          onChange={(v) => setRisk({ liquidationExposure: v as ExposureLevel })}
   608→        />
   609→        <RiskSelect
   610→          label="Smart contract risk"
   611→          value={risk.smartContractRisk}
   612→          options={riskOptions}
   613→          onChange={(v) => setRisk({ smartContractRisk: v as RiskLevel })}
   614→        />
   615→        <RiskSelect
   616→          label="Impermanent loss"
